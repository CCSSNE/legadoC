```kotlin
// BaseReadAloudService.kt
private var currentEngine: ReadAloudEngine? = null

private fun createEngine(book: Book): ReadAloudEngine {
    return when {
        book.isAudio -> SourceAudioEngine(this, engineCallback)
        ReadAloud.httpTTS != null -> HttpTtsEngine(this, engineCallback, httpTtsId)
        else -> SystemTtsEngine(this, engineCallback)
    }
}

private val engineCallback = object : ReadAloudEngine.Callback {
    override fun onProgressUpdate(position: Int, duration: Int) {
        postEvent(EventBus.READ_ALOUD_PROGRESS, Pair(position, duration))
    }
    override fun onCompletion() {
        // 播放下一章
    }
    override fun onError(error: String) {
        // 显示错误
    }
    override fun onLoading(loading: Boolean) {
        // 更新加载状态
    }
}
```
